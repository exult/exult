#	          dynamic_lists, list_layout

#	container_area: Defines the object placement area within gumps

#	special: Gumps that cannot be containers or lack close buttons
#	snap_zones: Optional guided placement zones for dynamic containers

#	dynamic_shapes: Data-driven shape displays, set by usecode
#	dynamic_sliders: Data-driven slider controls, set/read by usecode
#	dynamic_lists: Data-driven scrolling lists, from contents or usecode
#	list_layout: Layout of the container list view (Containers: List)
#

%%section snap_zones
# Defines guided placement zones for ritual containers.
# When an item is dropped in a zone, it automatically snaps to the target position.
#
# Format: :gump_shape/zone_id/zone_x/zone_y/zone_w/zone_h/snap_x/snap_y/priority/zone_type
#
# gump_shape = Shape number in gumps.vga
# zone_id    = Unique identifier for this zone (descriptive, e.g., "north_candle")
# zone_x/y   = Top-left corner of detection area (relative to container_area)
# zone_w/h   = Width/height of detection area (defines the snap zone size)
# snap_x/y   = Target position where items snap to (relative to container_area)
# priority   = Selection priority for overlapping zones (higher = checked first)
# zone_type  = Category label (e.g., "candle", "reagent", "focus") - informational only
#
# Example - Pentagram ritual container (uncomment and modify as needed):
# :111/north_candle/20/0/20/20/30/10/3/candle
# :111/south_candle/55/110/20/20/65/120/3/candle
# :111/center_focus/55/55/20/20/65/65/4/focus
#
%%endsection

%%section dynamic_buttons
# Format: :gump_shape/button_shape/pos_x/pos_y/default_frame/clicked_frame/sound_fx/usecode_fn/action_type/visibility_flag/usecode_param
#
# button_shape    = Shape number in gumps.vga for the button graphic
# pos_x/pos_y     = Button position relative to gump origin
# default_frame   = Frame displayed when idle (normally 0)
# clicked_frame   = Frame displayed when pushed (normally 1)
# sound_fx        = SFX index on click (-1 = none)
# usecode_fn      = Usecode function to call (-1 = none)
# action_type     = 0=nothing, 1=close gump (only when usecode_fn is -1; a button
#                   with a usecode callback must close via UI_close_gump2), 2=toggle pushed state
# visibility_flag = Global flag controlling visibility (-1 = always visible)
# usecode_param   = Legacy, read but unused (event is always double_click; use -1)
#
%%endsection

%%section dynamic_texts
# Format: :gump_shape/field_id/pos_x/pos_y/font_num[/halign[/valign]]
#
# field_id   = Unique integer ID for usecode to address this field
# pos_x/pos_y = Position relative to gump origin
# font_num   = Font number from fonts.vga (0=normal, 2=small, 5=tiny)
# halign     = Optional: 0=left (default, pos_x is left edge), 1=right (pos_x is
#              right edge), 2=center (pos_x is center)
# valign     = Optional: 0=top (default, pos_y is top edge), 1=bottom (pos_y is
#              bottom edge), 2=center (pos_y is middle). Needs halign.
#
# Text is set at runtime via UI_set_gump_text(item, field_id, "text")
#
%%endsection

%%section dynamic_shapes
# Format: :gump_shape/field_id/pos_x/pos_y/shape_num/frame_num/shapefile[/list_id[/box_w/box_h]]
#
# field_id      = Unique integer ID for usecode to address this field
# pos_x/pos_y   = Position relative to gump origin
# shape_num     = Initial shape number (0 = none until set by usecode)
# frame_num     = Initial frame number
# shapefile     = 0 = SF_SHAPES_VGA (shapes.vga), 1 = SF_GUMPS_VGA (gumps.vga)
# list_id       = Optional: show the selected row of the dynamic list with
#                 this field_id (-1 = none, the default)
# box_w/box_h   = Optional: pos_x/pos_y is a box's top-left; the shape is
#                 centred in it and enlarged up to 4x (0 = off, the default)
#
# Shape is set at runtime via UI_set_gump_shape(item, field_id, shape, frame)
#
%%endsection

%%section dynamic_sliders
# Format: :gump_shape/field_id/pos_x/pos_y/dec_shape/inc_shape/thumb_shape/min_val/max_val/step/default_val/extent/usecode_fn/orientation
#
# field_id      = Unique integer ID for usecode to address this slider
# pos_x/pos_y   = Position relative to gump origin
# dec_shape     = Shape number in gumps.vga for decrement arrow button
# inc_shape     = Shape number in gumps.vga for increment arrow button
# thumb_shape   = Shape number in gumps.vga for the draggable thumb
# min_val       = Minimum slider value
# max_val       = Maximum slider value
# step          = Value increment per click/key
# default_val   = Initial value when gump opens
# extent        = Pixel length of the slider track
# usecode_fn    = Usecode function called on value change (-1 = none)
# orientation   = 0 = vertical, 1 = horizontal
#
# Value is set/read at runtime via:
#   UI_set_slider_value(item, field_id, value)
#   UI_get_slider_value(item, field_id)
#
%%endsection

%%section dynamic_lists
# Format: :gump_shape/field_id/x/y/width/rows/row_height/font/source/up_shape/down_shape/thumb_shape/select_fn/activate_fn/rows_fn/flags
#
# field_id      = Unique integer ID for usecode to address this list
# x/y           = Top-left of the first row, relative to gump origin
# width         = Row width in pixels; the scroll controls sit to the right
# rows          = Visible rows
# row_height    = Pixel height of each row
# font          = Row font (fonts.vga)
# source        = 0 = the gump container's contents (drag in and out),
#                 1 = rows set by usecode
# up_shape      = gumps.vga scroll-up arrow (-1 = Exult's own scroll controls)
# down_shape    = gumps.vga scroll-down arrow
# thumb_shape   = gumps.vga thumb (-1 = spinner: arrows only)
# select_fn     = Usecode called when the player changes the selection (-1 = none)
# activate_fn   = Usecode called on double-click or Enter on a source 1 row (-1 = none)
# rows_fn       = Usecode called after source 0 rows changed (-1 = none)
# flags         = 0x01 = "N-M of T" line under the rows, 0x02 = "Empty" text
#
# Rows and selection at runtime (slots provisional), see docs/dynamic_lists.md:
#   UI_set_list_rows(item, field_id, rows)
#   UI_set_list_row(item, field_id, index, row)
#   UI_get_list_selection(item, field_id)
#   UI_set_list_selection(item, field_id, index)
#   UI_get_list_count(item, field_id)
#   UI_get_list_object(item, field_id, index)
#
# Example (not active): a contents list with Exult's scroll controls
# :<gump>/1/20/30/123/8/12/2/0/-1/-1/-1/-1/-1/-1/3
#
%%endsection

%%section list_layout
# Layout of the container list view, used only when the player sets
# Containers to List in Game Display Options.  See docs/dynamic_lists.md.
#
# Format: :gump_shape/part/fields...   (one line per part)
#
# gump_shape = the container's gump shape; -1 = the default for every
#              container gump.  A shape's lines replace those parts of
#              the default only.
# Numbers are decimal.  Positions are relative to the top-left of the
# widened background.
#
# Parts and their fields:
#   off                                  this gump opens in the picture
#                                        view (:-1/off: list view off)
#   background/shape/width/height/strip_x0/strip_x1/extra_w
#                                        gumps.vga art (-1 = gumps/scroll)
#                                        and its size (else the picture
#                                        view); columns strip_x0 to
#                                        strip_x1-1 repeat to widen it by
#                                        extra_w
#   list/<the 15 dynamic_lists fields after gump_shape>
#                                        the rows; source is always 0
#   title/x/y/width/font                 container name and rule
#   divider/x/y0/y1                      dotted line between the columns
#   panel/x/y/width/font/small_font      selected item's name and details
#   preview/x/y/box_w/box_h              selected item, enlarged in a box
#   picture/right/top                    the container's own picture, its
#                                        top-right corner
#   nav/left_x/right_x/y                 arrows through nested containers
#                                        (any value < 0 hides them)
#   usecode/open_fn                      usecode when the view opens
#                                        (item = the container; -1 = none)
#
# The list part's select_fn and rows_fn are the on-select and row-text
# hooks, as in dynamic_lists.
#
# Examples (not active):
# :<gump>/off
# :-1/list/0/51/42/123/8/12/2/0/-1/-1/-1/-1/-1/1234/3
#
# The default: Exult's built-in layout.
:-1/background/49/224/173/80/140/96
